SEERAAI — SOURCE-GROUNDED LEGAL ANSWER ENGINE
Version: 2.0
Date: 06 November 2026
Language support: Afaan Oromoo | አማርኛ | English

IMPORTANT INTEGRATION NOTE
--------------------------
This is the complete backend legal-answer engine and Express route module, not a
replacement for the entire SEERAAI application. It is designed to be added to
the existing React + TypeScript + Vite + Express + Firebase project.

It deliberately:
- Does NOT read or use the application's fixed FAQ collection.
- Does NOT delete or modify existing FAQ records.
- Answers the question submitted by the user at runtime.
- Retrieves only verified legal-source records from Firestore.
- Supports Ethiopian Federal and Oromia regional legislation and cassation decisions.
- Uses OpenAI Responses API first and Gemini as a fallback.
- Keeps API keys on the server.
- Refuses to invent article numbers, case numbers, holdings, or citations.
- Returns "insufficient verified sources" when the database does not contain
  adequate verified legal authority.

BEFORE DEPLOYMENT
-----------------
1. Back up the existing project.
2. Add the files below; do not replace the whole project.
3. Keep the existing Firebase project, database ID, rules, authentication,
   Storage, user roles, UI, and FAQ data.
4. Install/check dependencies:
   npm install express firebase-admin openai @google/genai zod
5. Set server environment variables:
   OPENAI_API_KEY
   GEMINI_API_KEY
   LEGAL_AI_PRIMARY=openai
   FIREBASE_PROJECT_ID
   FIRESTORE_DATABASE_ID
   LEGAL_AI_MODEL=gpt-5
   LEGAL_GEMINI_MODEL=gemini-2.5-flash
   LEGAL_ANSWER_MAX_SOURCES=12
6. Ensure the Firebase Admin SDK is initialized using the existing server-side
   service identity / Cloud Run service account. Do not put service-account
   secrets in frontend code.
7. Create and populate the Firestore collections described below.
8. Add the route to the existing server.ts as shown at the end.
9. Run tests in a staging environment before production deployment.

export type LegalLanguage = "om" | "am" | "en";
export type LegalJurisdiction = "federal" | "oromia" | "both" | "unknown";
export type LegalSourceType =
  | "proclamation"
  | "regulation"
  | "directive"
  | "civil_code"
  | "criminal_code"
  | "cassation_decision"
  | "court_decision"
  | "official_guideline"
  | "other";

export type VerificationStatus =
  | "VERIFIED"
  | "UNVERIFIED"
  | "REPEALED"
  | "AMENDED"
  | "HISTORICAL";

export interface LegalSourceRecord {
  id: string;
  title: string;
  titleOm?: string;
  titleAm?: string;
  titleEn?: string;

  sourceType: LegalSourceType;
  jurisdiction: "federal" | "oromia";
  issuingInstitution: string;

  proclamationNumber?: string;
  decisionNumber?: string;
  volume?: string;
  page?: string;
  articleNumbers?: string[];
  paragraphNumbers?: string[];

  issueDate?: string;
  effectiveDate?: string;
  repealDate?: string;
  status: VerificationStatus;

  legalIssueSummary: string;
  holdingOrRule: string;
  relevantText?: string;

  language?: LegalLanguage;
  searchTokens: string[];

  officialSourceName?: string;
  officialSourceUrl?: string;
  documentStoragePath?: string;

  verifiedBy?: string;
  verifiedAt?: string;
  lastReviewedAt?: string;

  // A source is not considered verified merely because this field is true.
  // The backend also requires status === "VERIFIED".
  sourceAuthenticityConfirmed?: boolean;
}

export interface LegalCitation {
  sourceId: string;
  title: string;
  sourceType: LegalSourceType;
  jurisdiction: "federal" | "oromia";
  issuingInstitution: string;
  proclamationNumber?: string;
  decisionNumber?: string;
  volume?: string;
  page?: string;
  articleNumbers?: string[];
  status: VerificationStatus;
  officialSourceName?: string;
  officialSourceUrl?: string;
}

export interface LegalAnswerRequest {
  question: string;
  language?: LegalLanguage;
  jurisdiction?: LegalJurisdiction;
  relevantDate?: string;
  facts?: string;
}

export interface LegalAnswerResponse {
  ok: boolean;
  answer: string;
  language: LegalLanguage;
  jurisdiction: LegalJurisdiction;
  citations: LegalCitation[];
  sourceCount: number;
  verification: "SOURCE_GROUNDED" | "INSUFFICIENT_SOURCES" | "AI_ERROR";
  disclaimer: string;
  requestId: string;
}

export interface RetrievedLegalSource extends LegalSourceRecord {
  score: number;
}

import type { Firestore } from "firebase-admin/firestore";
import {
  LegalJurisdiction,
  LegalSourceRecord,
  LegalSourceType,
  RetrievedLegalSource,
} from "./legalTypes";

const LAW_COLLECTION = "legal_sources";
const CASSATION_COLLECTION = "cassation_decisions";

const MAX_QUERY_TOKENS = 10;
const MAX_RESULTS_PER_COLLECTION = 30;

function normalizeText(value: string): string {
  return value
    .toLocaleLowerCase()
    .normalize("NFKC")
    .replace(/[^\p{L}\p{N}\s/-]/gu, " ")
    .replace(/\s+/g, " ")
    .trim();
}

function extractTokens(question: string): string[] {
  const stopWords = new Set([
    "the", "and", "for", "with", "from", "that", "this", "what", "when",
    "where", "which", "who", "how", "can", "may", "does", "into",
    "akka", "maal", "maaliif", "eessa", "yoom", "eenyu", "akkam",
    "እና", "ወይም", "ምን", "እንዴት", "ለምን",
  ]);

  return [...new Set(
    normalizeText(question)
      .split(" ")
      .map((token) => token.trim())
      .filter((token) => token.length >= 2 && !stopWords.has(token))
  )].slice(0, MAX_QUERY_TOKENS);
}

function jurisdictionMatches(
  sourceJurisdiction: string,
  requested: LegalJurisdiction
): boolean {
  if (requested === "unknown" || requested === "both") return true;
  return sourceJurisdiction === requested;
}

function sourceTypeMatchesCollection(
  source: Partial<LegalSourceRecord>,
  collection: "law" | "cassation"
): boolean {
  if (collection === "cassation") {
    return source.sourceType === "cassation_decision";
  }
  return source.sourceType !== "cassation_decision";
}

function scoreSource(
  source: LegalSourceRecord,
  tokens: string[]
): number {
  const searchable = normalizeText([
    source.title,
    source.titleOm,
    source.titleAm,
    source.titleEn,
    source.legalIssueSummary,
    source.holdingOrRule,
    source.relevantText,
    source.proclamationNumber,
    source.decisionNumber,
    ...(source.articleNumbers || []),
    ...(source.searchTokens || []),
  ].filter(Boolean).join(" "));

  let score = 0;
  for (const token of tokens) {
    if (searchable.includes(token)) score += 2;
    if ((source.searchTokens || []).some(
      (item) => normalizeText(item) === token
    )) score += 3;
    if (source.articleNumbers?.some(
      (item) => normalizeText(item) === token
    )) score += 4;
    if (source.decisionNumber &&
      normalizeText(source.decisionNumber).includes(token)) score += 5;
    if (source.proclamationNumber &&
      normalizeText(source.proclamationNumber).includes(token)) score += 5;
  }

  if (source.status === "VERIFIED" && source.sourceAuthenticityConfirmed) {
    score += 3;
  }
  return score;
}

function isUsableSource(source: LegalSourceRecord): boolean {
  return (
    source.status === "VERIFIED" &&
    source.sourceAuthenticityConfirmed === true &&
    Boolean(source.title) &&
    Boolean(source.issuingInstitution) &&
    Boolean(source.legalIssueSummary) &&
    Boolean(source.holdingOrRule) &&
    (source.sourceType !== "cassation_decision" ||
      Boolean(source.decisionNumber))
  );
}

function toSource(
  id: string,
  data: FirebaseFirestore.DocumentData
): LegalSourceRecord {
  return { id, ...data } as LegalSourceRecord;
}

export async function retrieveLegalSources(
  db: Firestore,
  question: string,
  requestedJurisdiction: LegalJurisdiction,
  maxSources = 12
): Promise<RetrievedLegalSource[]> {
  const tokens = extractTokens(question);
  if (!tokens.length) return [];

  // Firestore array-contains-any supports at most 10 values.
  const searchTokens = tokens.slice(0, 10);

  const [lawSnapshot, cassationSnapshot] = await Promise.all([
    db.collection(LAW_COLLECTION)
      .where("status", "==", "VERIFIED")
      .where("searchTokens", "array-contains-any", searchTokens)
      .limit(MAX_RESULTS_PER_COLLECTION)
      .get(),

    db.collection(CASSATION_COLLECTION)
      .where("status", "==", "VERIFIED")
      .where("searchTokens", "array-contains-any", searchTokens)
      .limit(MAX_RESULTS_PER_COLLECTION)
      .get(),
  ]);

  const candidates: LegalSourceRecord[] = [
    ...lawSnapshot.docs.map((doc) => toSource(doc.id, doc.data())),
    ...cassationSnapshot.docs.map((doc) => toSource(doc.id, doc.data())),
  ];

  const unique = new Map<string, RetrievedLegalSource>();

  for (const source of candidates) {
    if (!isUsableSource(source)) continue;
    if (!jurisdictionMatches(source.jurisdiction, requestedJurisdiction)) {
      continue;
    }
    if (
      source.sourceType !== "cassation_decision" &&
      source.jurisdiction !== "federal" &&
      source.jurisdiction !== "oromia"
    ) continue;

    const score = scoreSource(source, tokens);
    if (score <= 0) continue;
    unique.set(source.id, { ...source, score });
  }

  return [...unique.values()]
    .sort((a, b) => b.score - a.score)
    .slice(0, Math.min(Math.max(maxSources, 1), 20));
}

  return [
    `[SOURCE_ID: ${source.id}]`,
    `Title: ${source.title}`,
    `Source type: ${source.sourceType}`,
    `Jurisdiction: ${source.jurisdiction}`,
    `Issuing institution: ${source.issuingInstitution}`,
    `Proclamation number: ${source.proclamationNumber || "Not supplied"}`,
    `Decision number: ${source.decisionNumber || "Not supplied"}`,
    `Volume: ${source.volume || "Not supplied"}`,
    `Page: ${source.page || "Not supplied"}`,
    `Articles: ${(source.articleNumbers || []).join(", ") || "Not supplied"}`,
    `Status: ${source.status}`,
    `Issue summary: ${source.legalIssueSummary}`,
    `Rule / holding: ${source.holdingOrRule}`,
    `Relevant text: ${source.relevantText || "Not supplied"}`,
    `Official source: ${source.officialSourceName || "Not supplied"}`,
    `Official URL: ${source.officialSourceUrl || "Not supplied"}`,
  ].join("\n");
}

${args.facts || "No additional facts supplied."}

import OpenAI from "openai";
import { GoogleGenAI } from "@google/genai";
import type { Firestore } from "firebase-admin/firestore";
import {
  LegalAnswerRequest,
  LegalAnswerResponse,
  LegalCitation,
  LegalLanguage,
  LegalJurisdiction,
  RetrievedLegalSource,
} from "./legalTypes";
import { retrieveLegalSources } from "./legalSearch";
import { buildLegalSystemPrompt } from "./legalPrompt";

const openaiKey = process.env.OPENAI_API_KEY;
const geminiKey = process.env.GEMINI_API_KEY;

const openai = openaiKey ? new OpenAI({ apiKey: openaiKey }) : null;
const gemini = geminiKey ? new GoogleGenAI({ apiKey: geminiKey }) : null;

const OPENAI_MODEL = process.env.LEGAL_AI_MODEL || "gpt-5";
const GEMINI_MODEL =
  process.env.LEGAL_GEMINI_MODEL || "gemini-2.5-flash";

const MAX_QUESTION_LENGTH = 8000;
const MAX_FACTS_LENGTH = 12000;
const MIN_RELEVANT_SCORE = 2;

function detectLanguage(text: string): LegalLanguage {
  if (/[ሀ-ፐ]/u.test(text)) return "am";
  if (/[A-Za-z]/u.test(text) &&
      !/[Qq]or|[A-Za-z]/u.test(text)) return "en";
  // Afaan Oromoo is the default for Latin-script requests.
  return "om";
}

function normalizeJurisdiction(value?: LegalJurisdiction): LegalJurisdiction {
  if (
    value === "federal" ||
    value === "oromia" ||
    value === "both"
  ) return value;
  return "unknown";
}

function safeString(value: unknown, max: number): string {
  return typeof value === "string" ? value.trim().slice(0, max) : "";
}

function makeCitation(source: RetrievedLegalSource): LegalCitation {
  return {
    sourceId: source.id,
    title: source.title,
    sourceType: source.sourceType,
    jurisdiction: source.jurisdiction,
    issuingInstitution: source.issuingInstitution,
    proclamationNumber: source.proclamationNumber,
    decisionNumber: source.decisionNumber,
    volume: source.volume,
    page: source.page,
    articleNumbers: source.articleNumbers,
    status: source.status,
    officialSourceName: source.officialSourceName,
    officialSourceUrl: source.officialSourceUrl,
  };
}

function validCitedSourceIds(
  answer: string,
  sources: RetrievedLegalSource[]
): Set<string> {
  const allowed = new Set(sources.map((source) => source.id));
  const found = answer.matchAll(/\[SOURCE_ID:\s*([^\]]+)\]/g);
  const valid = new Set<string>();

  for (const match of found) {
    const id = match[1].trim();
    if (allowed.has(id)) valid.add(id);
  }
  return valid;
}

function removeInvalidCitationMarkers(
  answer: string,
  sources: RetrievedLegalSource[]
): string {
  const allowed = new Set(sources.map((source) => source.id));
  return answer.replace(
    /\[SOURCE_ID:\s*([^\]]+)\]/g,
    (whole, rawId: string) => {
      const id = String(rawId).trim();
      return allowed.has(id) ? whole : "";
    }
  );
}

function insufficientAnswer(language: LegalLanguage): string {
  if (language === "am") {
    return "በአሁኑ ጊዜ በተረጋገጠው የሕግ ዳታቤዝ ውስጥ አስተማማኝ የሕግ ውሳኔ ለመስጠት በቂ ተዛማጅ ምንጮች አልተገኙም። ተጨማሪ የሕግ ሰነድ፣ የአንቀጽ ጽሑፍ ወይም የፍርድ ውሳኔ ማረጋገጥ ያስፈልጋል።";
  }
  if (language === "en") {
    return "The available verified legal database does not contain enough relevant authority to provide a reliable legal conclusion. The applicable legal text, a verified cassation decision, or additional facts must be checked before reaching a conclusion.";
  }
  return "Kuusaan seeraa mirkanaa'e yeroo ammaa madda seeraa rogummaa qabu gahaa ta'e hin qabu. Kanaaf, murtii seeraa amanamaa kennuun hin danda'amu. Keewwata seeraa, murtii ijibbaataa mirkanaa'e ykn odeeffannoo dabalataa barbaachisu sakatta'uun dirqama.";
}

function disclaimer(language: LegalLanguage): string {
  if (language === "am") {
    return "ይህ ምላሽ ለሕግ ምርምርና መረጃ የተዘጋጀ ነው፤ የፍርድ ቤት ውሳኔ ወይም የግል የጠበቃ ምክር አይደለም። በጉዳዩ ላይ የሚያስፈልጉ ሰነዶችና የሕግ ምንጮች በባለሙያ መረጋገጥ አለባቸው።";
  }
  if (language === "en") {
    return "This response is for legal research and information only. It is not a court ruling or a substitute for professional legal advice. Verify the original legislation, case record, and facts before relying on it.";
  }
  return "Deebiin kun qorannoo fi odeeffannoo seeraa qofaaf kan qophaa'edha; murtii mana murtii ykn gorsa abukaatoo dhuunfaa bakka hin bu'u. Sanadoota seeraa, galmee murtii fi dhugaa dhimmaa madda jalqabaa irraa mirkaneessuun barbaachisaadha.";
}

async function askOpenAI(
  systemPrompt: string,
  question: string,
  facts: string
): Promise<string> {
  if (!openai) throw new Error("OPENAI_API_KEY is not configured.");

  const response = await openai.responses.create({
    model: OPENAI_MODEL,
    input: [
      {
        role: "system",
        content: [{ type: "input_text", text: systemPrompt }],
      },
      {
        role: "user",
        content: [{
          type: "input_text",
          text: `USER QUESTION:\n${question}\n\nADDITIONAL FACTS:\n${facts || "None"}`,
        }],
      },
    ],
  });

  const output = response.output_text?.trim();
  if (!output) throw new Error("OpenAI returned an empty answer.");
  return output;
}

async function askGemini(
  systemPrompt: string,
  question: string,
  facts: string
): Promise<string> {
  if (!gemini) throw new Error("GEMINI_API_KEY is not configured.");

  const response = await gemini.models.generateContent({
    model: GEMINI_MODEL,
    contents: [
      `${systemPrompt}\n\nUSER QUESTION:\n${question}\n\nADDITIONAL FACTS:\n${facts || "None"}`,
    ],
  });

  const output = response.text?.trim();
  if (!output) throw new Error("Gemini returned an empty answer.");
  return output;
}

async function generateWithFallback(
  systemPrompt: string,
  question: string,
  facts: string
): Promise<string> {
  const primary = (process.env.LEGAL_AI_PRIMARY || "openai").toLowerCase();

  if (primary === "gemini") {
    try {
      return await askGemini(systemPrompt, question, facts);
    } catch (geminiError) {
      if (!openai) throw geminiError;
      return askOpenAI(systemPrompt, question, facts);
    }
  }

  try {
    return await askOpenAI(systemPrompt, question, facts);
  } catch (openaiError) {
    if (!gemini) throw openaiError;
    return askGemini(systemPrompt, question, facts);
  }
}

export async function answerLegalQuestion(
  db: Firestore,
  input: LegalAnswerRequest,
  requestId: string
): Promise<LegalAnswerResponse> {
  const question = safeString(input.question, MAX_QUESTION_LENGTH);
  const facts = safeString(input.facts, MAX_FACTS_LENGTH);

  const language = input.language || detectLanguage(question);
  const jurisdiction = normalizeJurisdiction(input.jurisdiction);

  if (question.length < 4) {
    return {
      ok: false,
      answer: language === "en"
        ? "Please submit a complete legal question."
        : language === "am"
          ? "እባክዎ የተሟላ የሕግ ጥያቄ ያስገቡ።"
          : "Maaloo gaaffii seeraa guutuu galchi.",
      language,
      jurisdiction,
      citations: [],
      sourceCount: 0,
      verification: "INSUFFICIENT_SOURCES",
      disclaimer: disclaimer(language),
      requestId,
    };
  }

  const maxSources = Number(
    process.env.LEGAL_ANSWER_MAX_SOURCES || 12
  );

  const sources = await retrieveLegalSources(
    db,
    question,
    jurisdiction,
    Number.isFinite(maxSources) ? maxSources : 12
  );

  const relevantSources = sources.filter(
    (source) => source.score >= MIN_RELEVANT_SCORE
  );

  // No verified source = no generated legal conclusion.
  if (!relevantSources.length) {
    return {
      ok: true,
      answer: insufficientAnswer(language),
      language,
      jurisdiction,
      citations: [],
      sourceCount: 0,
      verification: "INSUFFICIENT_SOURCES",
      disclaimer: disclaimer(language),
      requestId,
    };
  }

  const systemPrompt = buildLegalSystemPrompt({
    language,
    jurisdiction,
    relevantDate: input.relevantDate,
    facts,
    sources: relevantSources,
  });

  try {
    const rawAnswer = await generateWithFallback(
      systemPrompt,
      question,
      facts
    );

    const answer = removeInvalidCitationMarkers(
      rawAnswer,
      relevantSources
    );

    const citedIds = validCitedSourceIds(answer, relevantSources);
    const citations = relevantSources
      .filter((source) => citedIds.has(source.id))
      .map(makeCitation);

    // If the model failed to cite any retrieved source, do not present its
    // answer as source-grounded legal advice.
    if (!citations.length) {
      return {
        ok: true,
        answer: insufficientAnswer(language),
        language,
        jurisdiction,
        citations: [],
        sourceCount: 0,
        verification: "INSUFFICIENT_SOURCES",
        disclaimer: disclaimer(language),
        requestId,
      };
    }

    return {
      ok: true,
      answer,
      language,
      jurisdiction,
      citations,
      sourceCount: citations.length,
      verification: "SOURCE_GROUNDED",
      disclaimer: disclaimer(language),
      requestId,
    };
  } catch (error) {
    // Do not expose API keys, raw provider responses, or internal stack traces.
    console.error("[SEERAAI legal-answer error]", {
      requestId,
      message: error instanceof Error ? error.message : "Unknown error",
    });

    return {
      ok: false,
      answer: language === "en"
        ? "The legal-answer service is temporarily unavailable. Please try again later."
        : language === "am"
          ? "የሕግ መልስ አገልግሎቱ ለጊዜው አይገኝም። እባክዎ ቆይተው ይሞክሩ።"
          : "Tajaajilli deebii seeraa yeroo ammaa hin hojjetu. Maaloo yeroo biraa irra deebi'ii yaali.",
      language,
      jurisdiction,
      citations: [],
      sourceCount: 0,
      verification: "AI_ERROR",
      disclaimer: disclaimer(language),
      requestId,
    };
  }
}

const router = Router();

const ALLOWED_LANGUAGES = new Set(["om", "am", "en"]);
const ALLOWED_JURISDICTIONS = new Set([
  "federal", "oromia", "both", "unknown",
]);

function parseRequest(body: unknown): LegalAnswerRequest | null {
  if (!body || typeof body !== "object") return null;
  const value = body as Record<string, unknown>;

  if (typeof value.question !== "string") return null;

  const language = typeof value.language === "string" &&
    ALLOWED_LANGUAGES.has(value.language)
      ? value.language as LegalLanguage
      : undefined;

  const jurisdiction = typeof value.jurisdiction === "string" &&
    ALLOWED_JURISDICTIONS.has(value.jurisdiction)
      ? value.jurisdiction as LegalJurisdiction
      : undefined;

  return {
    question: value.question,
    language,
    jurisdiction,
    relevantDate: typeof value.relevantDate === "string"
      ? value.relevantDate
      : undefined,
    facts: typeof value.facts === "string" ? value.facts : undefined,
  };
}

router.post("/ask", async (req: Request, res: Response) => {
  const requestId = randomUUID();

  try {
    // Verify Firebase ID token. Do not trust a user ID sent in request body.
    const authHeader = req.headers.authorization || "";
    const match = authHeader.match(/^Bearer\s+(.+)$/i);

    if (!match) {
      return res.status(401).json({
        ok: false,
        error: "AUTHENTICATION_REQUIRED",
        requestId,
      });
    }

    const decoded = await getAuth().verifyIdToken(match[1]);

    // Optional: enforce existing SEERAAI role rules here.
    // Do not replace the application's existing RBAC policy.
    // Example:
    // await assertUserMayUseLegalAI(decoded.uid);

    const parsed = parseRequest(req.body);
    if (!parsed) {
      return res.status(400).json({
        ok: false,
        error: "INVALID_REQUEST",
        requestId,
      });
    }

    if (parsed.question.trim().length < 4) {
      return res.status(400).json({
        ok: false,
        error: "QUESTION_TOO_SHORT",
        requestId,
      });
    }

    const db = getFirestore(
      process.env.FIRESTORE_DATABASE_ID || "(default)"
    );

    const result = await answerLegalQuestion(db, parsed, requestId);

    // Minimal audit event. Do not log confidential case facts or full prompts.
    try {
      await db.collection("legal_ai_audit").add({
        requestId,
        uid: decoded.uid,
        language: result.language,
        jurisdiction: result.jurisdiction,
        verification: result.verification,
        sourceIds: result.citations.map((item) => item.sourceId),
        createdAt: new Date().toISOString(),
      });
    } catch (auditError) {
      console.error("[SEERAAI audit write failed]", requestId);
    }

    return res.status(result.ok ? 200 : 503).json(result);
  } catch (error) {
    console.error("[SEERAAI route error]", {
      requestId,
      message: error instanceof Error ? error.message : "Unknown error",
    });

    return res.status(500).json({
      ok: false,
      error: "LEGAL_ANSWER_FAILED",
      requestId,
    });
  }
});

export default router;

import { getAuth } from "firebase/auth";
import type {
  LegalAnswerRequest,
  LegalAnswerResponse,
} from "../types/legalAnswer";

export async function askSEERAAILegalQuestion(
  input: LegalAnswerRequest
): Promise<LegalAnswerResponse> {
  const user = getAuth().currentUser;
  if (!user) throw new Error("Please sign in first.");

  const token = await user.getIdToken();

  const response = await fetch("/api/legal/ask", {
    method: "POST",
    headers: {
      "Content-Type": "application/json",
      Authorization: `Bearer ${token}`,
    },
    body: JSON.stringify(input),
  });

  const data = await response.json();

  if (!response.ok && !data?.answer) {
    throw new Error(data?.error || "Legal answer request failed.");
  }

  return data as LegalAnswerResponse;
}
