export type LegalDocumentType =
  | 'cassation_decision'
  | 'proclamation'
  | 'regulation'
  | 'directive'
  | 'unknown';

export type LegalJurisdiction = 'federal' | 'oromia' | 'unknown';

export type LegalCategory =
  | 'civil_contract'
  | 'labor'
  | 'property_land'
  | 'criminal'
  | 'family'
  | 'administrative'
  | 'procedure'
  | 'constitutional'
  | 'other'
  | 'unknown';

export type LocalizedText = {
  om?: string;
  am?: string;
  en?: string;
};

export interface ParsedTelegramLegalContent {
  id: string;
  messageId: number;
  channelHandle: string;
  postUrl: string;

  // Telegram irraa dhugumaan argame yoo kenname qofa guutama.
  // Guyyaa import godhame waliin wal hin makinaa.
  publishDate: string | null;
  ingestedAt: string;

  // Lakkoofsi wabii hin argamne yoo ta'e null dha; lakkoofsa hin uumamu.
  referenceNumber: string | null;
  type: LegalDocumentType;
  jurisdiction: LegalJurisdiction;
  category: LegalCategory;

  // Ramaddiin kun tilmaama jechoota irratti hundaa'a; ragaa seeraa mirkanaa'e miti.
  categoryConfidence: 'low' | 'medium' | 'unknown';

  // Hiikni/summariziin ogeessaan hin mirkanoofne yoo ta'e hin guutamin.
  title?: LocalizedText;
  summary?: LocalizedText;
  keyPrinciples?: LocalizedText;

  // Keewwata barruu keessaa baafame qofa; sirrii ta'uu isaa madda seeraa waliin mirkaneessi.
  keyArticles: string[];

  rawText: string;
  tags: string[];
  warnings: string[];
}

const CASE_REFERENCE_RE =
  /(?:lakk\.?\s*galmee|lakkoofsa\s*galmee|የ\s*ሰ[\s./-]*መ[\s./-]*ቁ\.?)[\s:#-]*([0-9][0-9\s/.-]*)/iu;

const PROCLAMATION_RE =
  /(?:\blabsii\b|\bproclamation\b|አዋጅ)\s*(?:(?:lakk\.?|lakkoofsa|no\.?)\s*|ቁጥር\s*)?([0-9]{1,5}\s*\/\s*[0-9]{2,4})/iu;

const REGULATION_RE =
  /(?:\bdambii\b|\bregulation\b|ደንብ)\s*(?:(?:lakk\.?|lakkoofsa|no\.?)\s*|ቁጥር\s*)?([0-9]{1,5}\s*\/\s*[0-9]{2,4})/iu;

const DIRECTIVE_RE =
  /(?:\bqajeelfama\b|\bdirective\b|መመሪያ)\s*(?:(?:lakk\.?|lakkoofsa|no\.?)\s*|ቁጥር\s*)?([0-9]{1,5}\s*\/\s*[0-9]{2,4})/iu;

const ARTICLE_RE =
  /(?:\bkeewwata\b|\bkeeyyata\b|\barticle\b|\bart\.?\b|አንቀጽ)\s*(\d+(?:\s*\(\s*(?:\d+|[a-z])\s*\))*)/giu;

const CATEGORY_TERMS: Record<Exclude<LegalCategory, 'other' | 'unknown'>, string[]> = {
  civil_contract: ['waliigaltee', 'contract', 'liiqii', 'debt', 'ውል'],
  labor: ['hojjetaa', 'hojii', 'employment', 'labour', 'labor', 'ሠራተኛ', 'የሥራ'],
  property_land: ['lafa', 'qabeenya', 'land', 'property', 'መሬት', 'ንብረት'],
  criminal: ['yakka', 'criminal', 'crime', 'offence', 'offense', 'ወንጀል'],
  family: ['maatii', 'gaa’ela', 'gaa\'ela', 'fuudhaa', 'heeruma', 'family', 'marriage', 'ቤተሰብ'],
  administrative: ['bulchiinsa', 'administrative', 'administration', 'አስተዳደር'],
  procedure: ['adeemsa falmii', 'civil procedure', 'criminal procedure', 'procedure', 'ሥነ ሥርዓት'],
  constitutional: ['heera mootummaa', 'constitution', 'constitutional', 'ሕገ መንግሥት'],
};

function normalizeNumber(value: string): string {
  return value.replace(/\s+/g, '').replace(/\.$/, '');
}

function toIsoDate(value?: Date | string | number): string | null {
  if (value === undefined || value === null) return null;
  const parsed = value instanceof Date ? value : new Date(value);
  if (!Number.isFinite(parsed.getTime())) {
    throw new Error('Telegram publish date sirrii miti.');
  }
  return parsed.toISOString();
}

function detectJurisdiction(header: string): LegalJurisdiction {
  const hasOromia =
    /naannoo\s+oromiyaa|oromia\s+national\s+regional\s+state|mana\s+murtii\s+waliigalaa\s+oromiyaa|supreme\s+court\s+of\s+oromia|caffee\s+oromiyaa/iu.test(header);

  const hasFederal =
    /federal\s+democratic\s+republic|federal\s+supreme\s+court|mana\s+murtii\s+waliigalaa\s+federaalaa| mootummaa\s+federaalaa|fdre/iu.test(header);

  if (hasOromia && !hasFederal) return 'oromia';
  if (hasFederal && !hasOromia) return 'federal';
  return 'unknown';
}

function detectCategory(text: string): {
  category: LegalCategory;
  confidence: 'low' | 'medium' | 'unknown';
} {
  const normalized = text.toLocaleLowerCase();
  const scores = Object.entries(CATEGORY_TERMS).map(([category, terms]) => ({
    category: category as Exclude<LegalCategory, 'other' | 'unknown'>,
    score: terms.reduce((total, term) => total + (normalized.includes(term) ? 1 : 0), 0),
  }));

  scores.sort((a, b) => b.score - a.score);
  const first = scores[0];
  const second = scores[1];

  if (!first || first.score === 0 || first.score === second?.score) {
    return { category: 'unknown', confidence: 'unknown' };
  }

  // Jechoota tokko irratti qofa hundaa'e amantaa gadi-aanaa qaba.
  return {
    category: first.category,
    confidence: first.score >= 2 ? 'medium' : 'low',
  };
}

function extractArticles(text: string): string[] {
  const found = new Set<string>();
  for (const match of text.matchAll(ARTICLE_RE)) {
    const articleNumber = match[1];
    if (articleNumber) found.add(articleNumber.replace(/\s+/g, ''));
  }
  return [...found];
}

export function parseTelegramLegalMessage(
  rawText: string,
  messageId: number,
  handle: string,
  telegramPublishDate?: Date | string | number,
): ParsedTelegramLegalContent {
  if (!Number.isSafeInteger(messageId) || messageId <= 0) {
    throw new Error('Telegram messageId lakkoofsa positive sirrii ta\'uu qaba.');
  }

  const channelHandle = handle.trim().replace(/^@/, '');
  if (!/^[A-Za-z0-9_]{1,64}$/.test(channelHandle)) {
    throw new Error('Telegram channel handle sirrii miti. @ fi qubee/lakkoofsa qofa fayyadami.');
  }

  const text = rawText.normalize('NFC').trim();
  if (!text) throw new Error('Telegram message duwwaa dha.');
  if (text.length > 1_000_000) {
    throw new Error('Telegram message baay\'ee dheeraa dha; gara systemitti galchuun dura qoodi.');
  }

  // Mata-duree/keeyyata jalqabaa irratti ramaddii goona; seera qaama barruu keessatti caqasameen
  // sanada guutuu akka Labsii/Dambii/Qajeelfamaatti dogoggoraan hin ramadnu.
  const header = text.split(/\r?\n/).slice(0, 10).join(' ').slice(0, 2000);

  const caseMatch = CASE_REFERENCE_RE.exec(header);
  const proclamationMatch = PROCLAMATION_RE.exec(header);
  const regulationMatch = REGULATION_RE.exec(header);
  const directiveMatch = DIRECTIVE_RE.exec(header);

  let type: LegalDocumentType = 'unknown';
  let referenceNumber: string | null = null;

  if (directiveMatch?.[1]) {
    type = 'directive';
    referenceNumber = `Qajeelfama ${normalizeNumber(directiveMatch[1])}`;
  } else if (regulationMatch?.[1]) {
    type = 'regulation';
    referenceNumber = `Dambii ${normalizeNumber(regulationMatch[1])}`;
  } else if (proclamationMatch?.[1]) {
    type = 'proclamation';
    referenceNumber = `Labsii ${normalizeNumber(proclamationMatch[1])}`;
  } else if (
    caseMatch?.[1] ||
    /dhaddacha\s+ijibbaataa|cassation\s+division|cassation\s+decision|የሰ\s*መ\s*ቁ/iu.test(header)
  ) {
    type = 'cassation_decision';
    referenceNumber = caseMatch?.[1]
      ? `Lakk. Galmee ${normalizeNumber(caseMatch[1])}`
      : null;
  }

  const jurisdiction = detectJurisdiction(header);
  const categoryResult = detectCategory(text);
  const publishDate = toIsoDate(telegramPublishDate);
  const keyArticles = extractArticles(text);
  const warnings: string[] = [];

  if (!publishDate) warnings.push('Guyyaan Telegram irratti maxxanfame hin kennamne; publishDate null dha.');
  if (!referenceNumber) warnings.push('Lakkoofsi wabii mata-duree irraa hin argamne; ofumaan hin uumamne.');
  if (type === 'unknown') warnings.push('Gosti sanadaa ifatti hin adda baane. Mirkaneessa namaatiin ramadi.');
  if (jurisdiction === 'unknown') warnings.push('Aangoon seeraa Federaalaa/Oromiyaa mata-duree irraa hin mirkanoofne.');
  if (categoryResult.confidence === 'low' || categoryResult.confidence === 'unknown') {
    warnings.push('Gosti dhimmaa tilmaama jechootaa qofa; ogeessa seeraatiin mirkaneessi.');
  }
  if (keyArticles.length === 0) {
    warnings.push('Keewwata lakkoofsa qabu parseriin hin argamne; keewwata hin uumne.');
  }

  return {
    // ID'n channel + messageId irratti hundaa'a; lakkoofsi galmee wal fakkaatus postoonni hin walitti bu'an.
    id: `tg_${channelHandle.toLowerCase()}_${messageId}`,
    messageId,
    channelHandle,
    postUrl: `https://t.me/${channelHandle}/${messageId}`,
    publishDate,
    ingestedAt: new Date().toISOString(),
    referenceNumber,
    type,
    jurisdiction,
    category: categoryResult.category,
    categoryConfidence: categoryResult.confidence,
    keyArticles,
    rawText: text,
    tags: [channelHandle, type, categoryResult.category, ...(referenceNumber ? [referenceNumber] : [])],
    warnings,
  };
}

3. FIRESTORE SECURITY RULES FOYYA'AN
------------------------------------
Koodiin armaan gadii claim Firebase Auth keessatti admin, legalContentManager, role fi institutionId jedhaman jiraachuu isaa irratti hundaa'a.

- admin: true ykn legalContentManager: true jechuun server/Admin SDK qofaan custom claim kenni.
- role: judge, assistant_judge, registrar, legal_expert ykn auditor keessaa tokko.
- institutionId: dhaabbata/ mana murtii fayyadamaan itti ramadame.
- Public repository keessatti publicationStatus = 'published' fi redacted = true ta'e qofa dubbifama.
- Firestore rules field tokko tokko qofa dhoksuu hin danda'an. Kanaaf rawText odeeffannoo dhuunfaa qabu public document keessa hin kaa'in; private collection keessatti kuusi, public copy qulqullaa'e qofa maxxansi.

-------------------- RULES START --------------------

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function signedIn() {
      return request.auth != null;
    }

    function isAdmin() {
      return signedIn() && request.auth.token.admin == true;
    }

    function isLegalRepositoryManager() {
      return signedIn() && (
        request.auth.token.admin == true ||
        request.auth.token.legalContentManager == true
      );
    }

    function hasCourtRole() {
      return signedIn() && request.auth.token.role in [
        'judge', 'assistant_judge', 'registrar', 'legal_expert', 'auditor'
      ];
    }

    function canEditCases() {
      return signedIn() && request.auth.token.role in [
        'judge', 'assistant_judge', 'registrar'
      ];
    }

    function sameInstitution(data) {
      return signedIn()
        && request.auth.token.institutionId != null
        && data.institutionId == request.auth.token.institutionId;
    }

    // Telegram legal repository: sanada maxxanfamee fi qulqullaa'e qofa public dha.
    match /telegram_legal_documents/{docId} {
      allow get, list: if resource.data.publicationStatus == 'published'
                       && resource.data.redacted == true;
      allow create, update, delete: if isLegalRepositoryManager();
    }

    // Official legislation repository: publish/review workflow qabaachuu qaba.
    match /legislation_repository/{docId} {
      allow get, list: if resource.data.publicationStatus == 'published'
                       && resource.data.redacted == true;
      allow create, update, delete: if isLegalRepositoryManager();
    }

    // User documents: abbaa galmee qofatu dubbisa/fooyyeessa/haqa.
    match /user_documents/{docId} {
      allow get, list: if signedIn() && resource.data.userId == request.auth.uid;

      allow create: if signedIn()
                    && request.resource.data.userId == request.auth.uid;

      allow update: if signedIn()
                    && resource.data.userId == request.auth.uid
                    && request.resource.data.userId == resource.data.userId;

      allow delete: if signedIn()
                    && resource.data.userId == request.auth.uid;
    }

    // Court cases: dhaabbata tokko keessatti hojjettoota aangoo qaban qofa.
    // Auditor dubbisuu danda'a; garuu galmee jijjiiruu hin danda'u.
    match /cases/{caseId} {
      allow get, list: if hasCourtRole() && sameInstitution(resource.data);

      allow create: if canEditCases()
                    && request.auth.token.institutionId != null
                    && request.resource.data.institutionId == request.auth.token.institutionId;

      allow update: if canEditCases()
                    && sameInstitution(resource.data)
                    && request.resource.data.institutionId == resource.data.institutionId;

      allow delete: if isAdmin() && sameInstitution(resource.data);
    }

    // Waan biraa hunda dhoowwi. Match biraa keessatti allow if jiraannaan,
    // Firestore rules keessatti allow tokko illee dhugaa ta'uun ni hayyama;
    // kanaaf paths hunda addaan sakatta'i.
    match /{document=**} {
      allow read, write: if false;
    }
  }
}

4. INGESTION / KUUSAA TELEGRAM – YAADA NAGEENYAA
------------------------------------------------
Parserii browser/client irraa kallattiin public collection irratti barreessisuun hin gorfamu. Ingestion server-side (Cloud Functions ykn Cloud Run) keessatti raawwadhu.

Flow gorfamu:
1. Telegram irraa message, messageId, channelHandle fi Telegram publish timestamp fudhadhu.
2. parseTelegramLegalMessage(...) fayyadami.
3. Raw source private collection keessatti kuusi; access isaa hojii seeraa aangoo qabu qofaan daangeessi.
4. Madda Telegram, link, lakkoofsa galmee/labsii fi guutummaa barruu namaan mirkaneessi.
5. Ragaa mootummaa/galmee murtii irraa mirkanaa'e, personal data irraa qulqullaa'e qofa public copy godhi.
6. Public document irratti publicationStatus='published' fi redacted=true kaa'i. Review hin xumuramne yoo ta'e 'draft' ykn 'pending_review' godhi.
7. createdAt, updatedAt, verifiedByUid, verifiedAt fi audit log server-side irratti galmeessi.

Telegram post link qofaan murtiin Dhaddacha Ijibbaataa dhugaa ta'uu hin mirkanaa'u. Link'n source tracing qofaaf gargaara; garuu ragaa original/official copy waliin mirkaneessuun barbaachisaa dha.
