// ===================================================== // 18. SEARCH
TEXT NORMALIZATION //
=====================================================

function normalizeSearchText(value: string): string { return value
.normalize(‘NFKC’) .toLocaleLowerCase() .replace(/+/g, ’ ’) .trim(); }

function extractSearchTerms(value: string): string[] { const normalized
= normalizeSearchText(value); const matches = normalized.match(/[]+/gu)
?? [];

return [ …new Set(matches.filter(term => term.length > 1)) ]; }

function flattenMultilingualText(value: MultilingualText): string {
return [value.om, value.am, value.en].join(’ ’); }

// ===================================================== // 19.
CASSATION SEARCHABLE CONTENT //
=====================================================

function getCassationSearchableText( record: CassationArchiveRecord ):
string { const principleTexts = record.bindingPrinciples.map(item =>
flattenMultilingualText(item.principle) );

const issueTexts = record.legalQuestions.map( flattenMultilingualText );

return [ flattenMultilingualText(record.subject), …issueTexts,
flattenMultilingualText(record.facts),
flattenMultilingualText(record.courtReasoning),
flattenMultilingualText(record.decisionOutcome),
flattenMultilingualText(record.fullText), …principleTexts,
…record.applicableLaws, …record.applicableArticles, …record.tags,
record.caseNumber ].join(’ ’); }

// ===================================================== // 20. MATCHED
LEGAL ISSUES // =====================================================

function findMatchedIssues( record: CassationArchiveRecord, queryTerms:
string[] ): string[] { return record.legalQuestions .filter(issue => {
const issueTerms = new Set(
extractSearchTerms(flattenMultilingualText(issue)) );

      return queryTerms.some(term => issueTerms.has(term));
    })
    .map(issue => flattenMultilingualText(issue));

}

// ===================================================== // 21. MATCHED
LEGAL ARTICLES // =====================================================

function normalizeLegalReference(value: string): string { return
normalizeSearchText(value) .replace(/[^\p{L}\p{N}]/gu, ’’); }

function findMatchedArticles( record: CassationArchiveRecord,
requestedArticles?: string[] ): string[] { if
(!requestedArticles?.length) return [];

const normalizedRequested = new Set(
requestedArticles.map(normalizeLegalReference) );

return record.applicableArticles.filter(article =>
normalizedRequested.has(normalizeLegalReference(article)) ); }

// ===================================================== // 22.
CASSATION RELEVANCE SCORING //
=====================================================

export function calculateCassationRelevance( record:
CassationArchiveRecord, request: CassationSearchRequest ): number { let
score = 0;

if ( request.caseNumber && normalizeLegalReference(record.caseNumber)
=== normalizeLegalReference(request.caseNumber) ) { score += 40; }

if ( request.jurisdiction && record.jurisdiction ===
request.jurisdiction ) { score += 10; }

if ( request.volume && String(record.volume ?? ’’) === request.volume )
{ score += 10; }

if (request.yearEC && record.yearEC === request.yearEC) { score += 8; }

if (request.yearGC && record.yearGC === request.yearGC) { score += 8; }

const matchedArticles = findMatchedArticles( record,
request.applicableArticles );

score += Math.min(18, matchedArticles.length * 6);

const queryTerms = extractSearchTerms(request.question); const
searchableTerms = new Set(
extractSearchTerms(getCassationSearchableText(record)) );

const matchedTerms = queryTerms.filter(term => searchableTerms.has(term)
);

score += Math.min(25, matchedTerms.length * 5);

if (record.verification.documentVerified) score += 2; if
(record.verification.metadataVerified) score += 2; if
(record.verification.textVerified) score += 1; if
(record.verification.bindingStatusVerified) score += 1;

return Math.min(100, score); }

// ===================================================== // 23. SOURCE
VERIFICATION CHECK //
=====================================================

function isArchiveRecordVerified( record: CassationArchiveRecord,
liveSourceVerified: boolean ): boolean { return ( record.verified ===
true && record.verification.documentVerified === true &&
record.verification.textVerified === true &&
record.verification.metadataVerified === true &&
record.verification.bindingStatusVerified === true && liveSourceVerified
=== true ); }

// ===================================================== // 24.
RELEVANCE EXPLANATION //
=====================================================

function buildRelevanceExplanation( record: CassationArchiveRecord,
request: CassationSearchRequest, matchedTerms: string[],
matchedArticles: string[] ): string { const reasons: string[] = [];

if ( request.caseNumber && normalizeLegalReference(record.caseNumber)
=== normalizeLegalReference(request.caseNumber) ) {
reasons.push(‘Lakkoofsi galmee sirriitti wal sima.’); }

if ( request.jurisdiction && record.jurisdiction ===
request.jurisdiction ) { reasons.push(‘Aangoon seeraa wal sima.’); }

if (matchedArticles.length > 0) { reasons.push(‘Keewwatni seeraa wal
siman argamaniiru.’); }

if (matchedTerms.length > 0) { reasons.push(
${matchedTerms.length} jechoonni gaaffii waliin wal siman argamaniiru.
); }

if (reasons.length === 0) { return ( ‘Bu’aan kun sirna barbaacha keessaa
argame;’ + ‘rogummaa isaa ragaa dabalataa irratti hundaa’uun madaaluun
barbaachisa.’ ); }

return reasons.join(’ ’); }

// ===================================================== // 25. MAIN
TELEGRAM CASSATION ARCHIVE SEARCH //
=====================================================

export async function searchCassationArchive( repository:
CassationArchiveRepository, request: CassationSearchRequest ): Promise {
const records = await repository.searchByText( request.question, request
);

const sourceVerificationResults = await Promise.all( records.map(async
record => { try { return await repository.verifySource(record.id); }
catch { return false; } }) );

const ranked: CassationSearchResult[] = [];

for (let index = 0; index < records.length; index++) { const record =
records[index]; const liveSourceVerified =
sourceVerificationResults[index];

    const relevanceScore = calculateCassationRelevance(
      record,
      request
    );

    const queryTerms = extractSearchTerms(request.question);
    const searchableTerms = new Set(
      extractSearchTerms(getCassationSearchableText(record))
    );

    const matchedTerms = queryTerms.filter(term =>
      searchableTerms.has(term)
    );

    const matchedIssues = findMatchedIssues(
      record,
      queryTerms
    );

    const matchedArticles = findMatchedArticles(
      record,
      request.applicableArticles
    );

    const sourceVerified = isArchiveRecordVerified(
      record,
      liveSourceVerified
    );

    ranked.push({
      record,
      relevanceScore,
      matchedIssues,
      matchedArticles,
      matchedTerms,
      relevanceExplanation: buildRelevanceExplanation(
        record,
        request,
        matchedTerms,
        matchedArticles
      ),
      sourceVerified,
      citation: {
        court: record.court,
        caseNumber: record.caseNumber,
        volume:
          record.volume === undefined
            ? undefined
            : String(record.volume),
        decisionDate: record.decisionDate,
        telegramMessageId: record.telegramSource.messageId,
        sourceLink: record.telegramSource.messageLink
      }
    });

}

const sorted = ranked.sort( (a, b) => b.relevanceScore -
a.relevanceScore );

const relevant = sorted.filter( item => item.relevanceScore >= 15 );

const exactMatches = request.caseNumber ? relevant.filter(item =>
normalizeLegalReference(item.record.caseNumber) ===
normalizeLegalReference(request.caseNumber!) ) : [];

const relatedDecisions = relevant.filter( item =>
!exactMatches.includes(item) );

const limit = Math.max( 1, Math.min(Math.floor(request.limit ?? 10),
100) );

const limitedResults = relevant.slice(0, limit);

const hasRelevantResults = limitedResults.length > 0; const
hasVerifiedResult = limitedResults.some( item => item.sourceVerified );

const insufficientEvidence = !hasRelevantResults || !hasVerifiedResult;

let warning: string | undefined;

if (!hasRelevantResults) { warning = ‘Murtii rogummaa qabu hin
argamne.’ + ‘Gaaffii, lakkoofsa galmee ykn keewwata seeraa sirreessuun
barbaachisaa ta’uu danda’a.’; } else if (!hasVerifiedResult) { warning =
‘Murtiiwwan barbaachisoo ta’an argamaniiru;’ + ‘garuu madda isaanii
guutummaatti hin mirkanoofne.’ + ‘Murtiiwwan kana akka madda seeraa
mirkanaa’eetti fayyadamuun dura ragaa jalqabaa sakatta’i.’; }

const searchedJurisdictions: Jurisdiction[] = request.jurisdiction ?
[request.jurisdiction] : […new Set(records.map(record =>
record.jurisdiction))];

if (searchedJurisdictions.length === 0) {
searchedJurisdictions.push(‘oromia’, ‘federal’); }

return { query: request.question, totalResults: records.length,
relevantResults: limitedResults, exactMatches, relatedDecisions,
insufficientEvidence, warning, searchedJurisdictions, searchedAt: new
Date().toISOString() }; }
