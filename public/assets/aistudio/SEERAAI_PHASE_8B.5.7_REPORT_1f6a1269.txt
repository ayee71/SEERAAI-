--- BEGIN .github/workflows/rules-test.yml ---
name: rules-emulator-tests

on:
  workflow_dispatch:

permissions:
  contents: read

jobs:
  rules-tests:
    runs-on: ubuntu-latest
    timeout-minutes: 20
    env:
      GCLOUD_PROJECT: demo-seeraai-test
      FIRESTORE_CANDIDATE: firestore.rules.phase8b5.3.integrated.candidate
      STORAGE_CANDIDATE: storage.rules.phase8b5.4.integrated.candidate
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: '17'

      - uses: actions/setup-node@v4
        with:
          node-version: '20'
          cache: npm

      - name: Install dependencies from lockfile
        run: npm ci

      - name: Guard - isolated config only
        run: |
          set -euo pipefail
          test -f "$FIRESTORE_CANDIDATE"
          test -f "$STORAGE_CANDIDATE"
          test -f firebase.emulator.test.json
          grep -q "$FIRESTORE_CANDIDATE" firebase.emulator.test.json
          grep -q "$STORAGE_CANDIDATE" firebase.emulator.test.json
          if grep -rq "leafy-rush-xvd6f" firebase.emulator.test.json .github test 2>/dev/null; then
            echo "Production project ID found in test setup"; exit 1
          fi

      - name: Prepare evidence folder and hashes (before)
        run: |
          mkdir -p evidence
          sha256sum "$FIRESTORE_CANDIDATE" "$STORAGE_CANDIDATE" | tee evidence/hashes_before.txt
          wc -lc "$FIRESTORE_CANDIDATE" "$STORAGE_CANDIDATE" | tee evidence/sizes.txt
          npx --no-install firebase --version | tee evidence/tool_versions.txt
          java -version 2>> evidence/tool_versions.txt
          node -v >> evidence/tool_versions.txt

      - name: Run Firestore + Storage rules tests in emulators
        run: |
          set +e
          npx --no-install firebase emulators:exec \
            --config firebase.emulator.test.json \
            --project demo-seeraai-test \
            --only firestore,storage \
            "npm test" 2>&1 | tee evidence/test_output.txt
          code=${PIPESTATUS[0]}
          echo "$code" > evidence/exit_code.txt
          exit 0

      - name: Hashes (after) and unchanged check
        if: always()
        run: |
          sha256sum "$FIRESTORE_CANDIDATE" "$STORAGE_CANDIDATE" | tee evidence/hashes_after.txt
          diff evidence/hashes_before.txt evidence/hashes_after.txt \
            && echo "HASHES UNCHANGED" | tee evidence/hash_check.txt \
            || { echo "HASHES CHANGED" | tee evidence/hash_check.txt; exit 1; }
          cp -f firebase-debug.log firestore-debug.log storage-debug.log ui-debug.log evidence/ 2>/dev/null || true

      - name: Upload evidence
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: rules-test-evidence
          path: evidence/
          if-no-files-found: error

      - name: Fail job if tests failed
        run: |
          code=$(cat evidence/exit_code.txt)
          echo "Test exit code: $code"
          test "$code" = "0"
--- END ---

Why the run step uses "exit 0": it lets the evidence/hash/upload steps
run, and the LAST step fails the job with the real exit code. A failed
test therefore still produces a red workflow AND downloadable evidence.
