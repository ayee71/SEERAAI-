export type RecordingState = "idle" | "recording" | "paused" | "stopped";

export interface AudioRecordingResult {
  blob: Blob;
  mimeType: string;
  durationSeconds: number;
  sizeBytes: number;
}

export interface CourtroomAudioCallbacks {
  onStateChange?: (state: RecordingState) => void;
  onDurationChange?: (seconds: number) => void;
  onVolumeChange?: (dbFS: number) => void;
  onError?: (error: Error) => void;
}

export interface CourtroomAudioOptions {
  timesliceMs?: number;
  audioBitsPerSecond?: number;
  audioConstraints?: MediaTrackConstraints;
  callbacks?: CourtroomAudioCallbacks;
}

type BrowserWindow = Window & {
  webkitAudioContext?: typeof AudioContext;
};

export class CourtroomAudioService {
  private stream: MediaStream | null = null;
  private recorder: MediaRecorder | null = null;
  private chunks: Blob[] = [];
  private audioContext: AudioContext | null = null;
  private analyser: AnalyserNode | null = null;
  private source: MediaStreamAudioSourceNode | null = null;

  private timerId: ReturnType<typeof setInterval> | null = null;
  private volumeFrameId: number | null = null;
  private elapsedMs = 0;
  private segmentStartedAt: number | null = null;

  private stopPromise: Promise<AudioRecordingResult | null> | null = null;
  private resolveStop:
    | ((result: AudioRecordingResult | null) => void)
    | null = null;
  private rejectStop: ((error: Error) => void) | null = null;

  private callbacks: CourtroomAudioCallbacks;
  private readonly options: CourtroomAudioOptions;
  private disposed = false;

  public state: RecordingState = "idle";
  public durationSeconds = 0;

  constructor(options: CourtroomAudioOptions = {}) {
    this.options = {
      timesliceMs: 1000,
      audioBitsPerSecond: 128000,
      ...options,
    };
    this.callbacks = options.callbacks ?? {};
  }

  public setCallbacks(callbacks: CourtroomAudioCallbacks): void {
    this.callbacks = callbacks;
  }

  private setState(state: RecordingState): void {
    this.state = state;
    this.callbacks.onStateChange?.(state);
  }

  private toError(value: unknown): Error {
    return value instanceof Error ? value : new Error(String(value));
  }

  private reportError(value: unknown): Error {
    const error = this.toError(value);
    this.callbacks.onError?.(error);
    return error;
  }

  private chooseMimeType(): string | undefined {
    if (typeof MediaRecorder === "undefined") return undefined;
    const candidates = [
      "audio/webm;codecs=opus",
      "audio/mp4",
      "audio/ogg;codecs=opus",
      "audio/webm",
    ];
    return candidates.find((type) => MediaRecorder.isTypeSupported(type));
  }

  private async setupAnalyser(stream: MediaStream): Promise<void> {
    const win = window as BrowserWindow;
    const AudioContextCtor = win.AudioContext ?? win.webkitAudioContext;
    if (!AudioContextCtor) return;

    try {
      const context = new AudioContextCtor();
      this.audioContext = context;
      if (context.state === "suspended") await context.resume();

      const source = context.createMediaStreamSource(stream);
      const analyser = context.createAnalyser();
      analyser.fftSize = 2048;
      analyser.smoothingTimeConstant = 0.75;
      source.connect(analyser);

      this.source = source;
      this.analyser = analyser;
    } catch {
      await this.closeAudioContext();
      // Audio metering is optional; the recording can still continue.
    }
  }

  public async startRecording(): Promise<void> {
    if (this.disposed) throw new Error("Audio service has been disposed.");
    if (this.state === "recording" || this.state === "paused") {
      throw new Error("Waraabbiin duraan adeemaa jira.");
    }
    if (
      typeof window === "undefined" ||
      !window.isSecureContext ||
      !navigator.mediaDevices?.getUserMedia
    ) {
      throw new Error(
        "Microphone fayyadamuuf HTTPS fi browser deeggaru barbaachisa.",
      );
    }
    if (typeof MediaRecorder === "undefined") {
      throw new Error("Browser kun MediaRecorder hin deeggaru.");
    }

    await this.releaseResources();
    this.chunks = [];
    this.elapsedMs = 0;
    this.durationSeconds = 0;
    this.segmentStartedAt = null;

    try {
      const stream = await navigator.mediaDevices.getUserMedia({
        audio: this.options.audioConstraints ?? {
          echoCancellation: true,
          noiseSuppression: true,
          autoGainControl: true,
        },
      });
      this.stream = stream;
      await this.setupAnalyser(stream);

      const mimeType = this.chooseMimeType();
      const recorderOptions: MediaRecorderOptions = {
        audioBitsPerSecond: this.options.audioBitsPerSecond ?? 128000,
      };
      if (mimeType) recorderOptions.mimeType = mimeType;

      const recorder = new MediaRecorder(stream, recorderOptions);
      this.recorder = recorder;

      this.stopPromise = new Promise<AudioRecordingResult | null>(
        (resolve, reject) => {
          this.resolveStop = resolve;
          this.rejectStop = reject;
        },
      );

      recorder.addEventListener("dataavailable", (event: BlobEvent) => {
        if (event.data && event.data.size > 0) this.chunks.push(event.data);
      });

      recorder.addEventListener(
        "error",
        (event: Event) => {
          const domEvent = event as Event & { error?: DOMException };
          const error = this.reportError(
            domEvent.error ?? new Error("MediaRecorder error."),
          );
          // Stop event is still responsible for finalizing the pending promise.
          if (recorder.state !== "inactive") {
            try {
              recorder.stop();
            } catch {
              this.rejectStop?.(error);
            }
          } else {
            this.rejectStop?.(error);
          }
        },
        { once: true },
      );

      recorder.addEventListener(
        "stop",
        () => {
          void this.finishRecording(recorder);
        },
        { once: true },
      );

      stream.getAudioTracks().forEach((track) => {
        track.addEventListener(
          "ended",
          () => {
            if (this.state === "recording" || this.state === "paused") {
              this.reportError(new Error("Microphone connection addaan cite."));
              void this.stopRecording();
            }
          },
          { once: true },
        );
      });

      recorder.start(Math.max(250, this.options.timesliceMs ?? 1000));
      this.segmentStartedAt = performance.now();
      this.setState("recording");
      this.startTimer();
      this.startVolumeMeter();
    } catch (error) {
      await this.releaseResources();
      this.recorder = null;
      this.resolveStop?.(null);
      this.clearStopHandlers();
      this.setState("idle");
      throw this.reportError(error);
    }
  }

  public pauseRecording(): void {
    if (!this.recorder || this.state !== "recording") return;
    try {
      this.recorder.pause();
      this.closeActiveSegment();
      this.stopTimer();
      this.stopVolumeMeter();
      this.setState("paused");
    } catch (error) {
      this.reportError(error);
    }
  }

  public resumeRecording(): void {
    if (!this.recorder || this.state !== "paused") return;
    try {
      this.recorder.resume();
      this.segmentStartedAt = performance.now();
      this.setState("recording");
      this.startTimer();
      this.startVolumeMeter();
    } catch (error) {
      this.reportError(error);
    }
  }

  public stopRecording(): Promise<AudioRecordingResult | null> {
    if (this.stopPromise) {
      if (this.recorder && this.recorder.state !== "inactive") {
        this.closeActiveSegment();
        this.stopTimer();
        this.stopVolumeMeter();
        try {
          this.recorder.stop();
        } catch (error) {
          this.rejectStop?.(this.toError(error));
        }
      }
      return this.stopPromise;
    }
    return Promise.resolve(null);
  }

  private async finishRecording(recorder: MediaRecorder): Promise<void> {
    // Ignore stale stop events from a recorder replaced by a later session.
    if (this.recorder !== recorder) return;

    this.closeActiveSegment();
    this.stopTimer();
    this.stopVolumeMeter();

    const mimeType =
      recorder.mimeType || this.chunks[0]?.type || "application/octet-stream";
    const blob =
      this.chunks.length > 0
        ? new Blob(this.chunks, { type: mimeType })
        : null;

    await this.releaseResources();
    this.recorder = null;

    const result: AudioRecordingResult | null = blob
      ? {
          blob,
          mimeType,
          durationSeconds: this.durationSeconds,
          sizeBytes: blob.size,
        }
      : null;

    this.setState("stopped");
    this.resolveStop?.(result);
    this.clearStopHandlers();
  }

  private clearStopHandlers(): void {
    this.resolveStop = null;
    this.rejectStop = null;
    this.stopPromise = null;
  }

  private closeActiveSegment(): void {
    if (this.segmentStartedAt !== null) {
      this.elapsedMs += performance.now() - this.segmentStartedAt;
      this.segmentStartedAt = null;
    }
    this.updateDuration();
  }

  private updateDuration(): void {
    const activeMs =
      this.segmentStartedAt === null
        ? 0
        : performance.now() - this.segmentStartedAt;
    const seconds = Math.floor((this.elapsedMs + activeMs) / 1000);
    if (seconds !== this.durationSeconds) {
      this.durationSeconds = seconds;
      this.callbacks.onDurationChange?.(seconds);
    }
  }

  private startTimer(): void {
    this.stopTimer();
    this.timerId = setInterval(() => this.updateDuration(), 250);
  }

  private stopTimer(): void {
    if (this.timerId !== null) clearInterval(this.timerId);
    this.timerId = null;
  }

  public getVolumeDbFS(): number {
    if (!this.analyser) return -60;
    const samples = new Float32Array(this.analyser.fftSize);
    this.analyser.getFloatTimeDomainData(samples);

    let sumSquares = 0;
    for (const sample of samples) sumSquares += sample * sample;
    const rms = Math.sqrt(sumSquares / samples.length);
    const db = 20 * Math.log10(Math.max(rms, 1e-6));
    return Math.max(-60, Math.min(0, db));
  }

  private startVolumeMeter(): void {
    if (!this.analyser) return;
    this.stopVolumeMeter();

    const tick = () => {
      if (!this.analyser || this.state !== "recording") return;
      this.callbacks.onVolumeChange?.(this.getVolumeDbFS());
      this.volumeFrameId = requestAnimationFrame(tick);
    };
    this.volumeFrameId = requestAnimationFrame(tick);
  }

  private stopVolumeMeter(): void {
    if (this.volumeFrameId !== null) {
      cancelAnimationFrame(this.volumeFrameId);
    }
    this.volumeFrameId = null;
  }

  private async closeAudioContext(): Promise<void> {
    this.source?.disconnect();
    this.source = null;
    this.analyser = null;

    const context = this.audioContext;
    this.audioContext = null;
    if (context && context.state !== "closed") {
      try {
        await context.close();
      } catch {
        // Context may already have been closed by the browser.
      }
    }
  }

  private async releaseResources(): Promise<void> {
    this.stopTimer();
    this.stopVolumeMeter();

    const stream = this.stream;
    this.stream = null;
    stream?.getTracks().forEach((track) => track.stop());

    await this.closeAudioContext();
  }

  public async dispose(): Promise<void> {
    if (this.disposed) return;
    if (this.recorder && this.recorder.state !== "inactive") {
      await this.stopRecording();
    } else if (this.stopPromise) {
      await this.stopPromise.catch(() => null);
    } else {
      await this.releaseResources();
    }
    this.disposed = true;
  }
}

import { useCallback, useEffect, useRef, useState } from "react";
import {
  AudioRecordingResult,
  CourtroomAudioService,
  RecordingState,
} from "../services/CourtroomAudioService";

export interface CourtroomRecorderHook {
  state: RecordingState;
  durationSeconds: number;
  volumeDbFS: number;
  error: string | null;
  result: AudioRecordingResult | null;
  audioUrl: string | null;
  start: () => Promise<void>;
  pause: () => void;
  resume: () => void;
  stop: () => Promise<AudioRecordingResult | null>;
  clear: () => void;
}

export function useCourtroomRecorder(): CourtroomRecorderHook {
  const serviceRef = useRef<CourtroomAudioService | null>(null);
  const audioUrlRef = useRef<string | null>(null);

  const [state, setState] = useState<RecordingState>("idle");
  const [durationSeconds, setDurationSeconds] = useState(0);
  const [volumeDbFS, setVolumeDbFS] = useState(-60);
  const [error, setError] = useState<string | null>(null);
  const [result, setResult] = useState<AudioRecordingResult | null>(null);
  const [audioUrl, setAudioUrl] = useState<string | null>(null);

  useEffect(() => {
    const service = new CourtroomAudioService({
      callbacks: {
        onStateChange: setState,
        onDurationChange: setDurationSeconds,
        onVolumeChange: setVolumeDbFS,
        onError: (err) => setError(err.message),
      },
    });
    serviceRef.current = service;

    return () => {
      void service.dispose();
      if (audioUrlRef.current) URL.revokeObjectURL(audioUrlRef.current);
      audioUrlRef.current = null;
    };
  }, []);

  const clear = useCallback(() => {
    if (audioUrlRef.current) URL.revokeObjectURL(audioUrlRef.current);
    audioUrlRef.current = null;
    setAudioUrl(null);
    setResult(null);
    setError(null);
    setDurationSeconds(0);
    setVolumeDbFS(-60);
    setState("idle");
  }, []);

  const start = useCallback(async () => {
    setError(null);
    setResult(null);
    if (audioUrlRef.current) URL.revokeObjectURL(audioUrlRef.current);
    audioUrlRef.current = null;
    setAudioUrl(null);

    try {
      await serviceRef.current?.startRecording();
    } catch (err) {
      setError(err instanceof Error ? err.message : "Recording failed.");
    }
  }, []);

  const pause = useCallback(() => serviceRef.current?.pauseRecording(), []);
  const resume = useCallback(() => serviceRef.current?.resumeRecording(), []);

  const stop = useCallback(async () => {
    try {
      const recording = await serviceRef.current?.stopRecording() ?? null;
      if (recording) {
        setResult(recording);
        const url = URL.createObjectURL(recording.blob);
        if (audioUrlRef.current) URL.revokeObjectURL(audioUrlRef.current);
        audioUrlRef.current = url;
        setAudioUrl(url);
      }
      return recording;
    } catch (err) {
      setError(err instanceof Error ? err.message : "Could not stop recording.");
      return null;
    }
  }, []);

  return {
    state,
    durationSeconds,
    volumeDbFS,
    error,
    result,
    audioUrl,
    start,
    pause,
    resume,
    stop,
    clear,
  };
}

import type { NextFunction, Request, Response } from "express";
import { getAuth } from "firebase-admin/auth";

export interface CourtUser {
  uid: string;
  institutionId: string;
  role: string;
  email?: string;
}

declare global {
  namespace Express {
    interface Request {
      courtUser?: CourtUser;
    }
  }
}

/**
 * Verify Firebase ID token. This proves identity only.
 * Authorization for institution/case access must also be performed.
 */
export async function requireCourtUser(
  req: Request,
  res: Response,
  next: NextFunction,
): Promise<void> {
  try {
    const header = req.headers.authorization;
    if (!header?.startsWith("Bearer ")) {
      res.status(401).json({ error: "Authentication required." });
      return;
    }

    const token = header.slice("Bearer ".length).trim();
    if (!token) {
      res.status(401).json({ error: "Authentication required." });
      return;
    }

    const decoded = await getAuth().verifyIdToken(token, true);
    const institutionId =
      typeof decoded.institutionId === "string" ? decoded.institutionId : "";
    const role = typeof decoded.role === "string" ? decoded.role : "";

    if (!institutionId || !role) {
      res.status(403).json({ error: "Institution membership is required." });
      return;
    }

    req.courtUser = {
      uid: decoded.uid,
      institutionId,
      role,
      email: decoded.email,
    };
    next();
  } catch {
    res.status(401).json({ error: "Invalid or expired authentication token." });
  }
}

const MAX_AUDIO_BYTES = 15 * 1024 * 1024;

function decodeBase64Audio(input: string): Buffer {
  const stripped = input.replace(
    /^data:audio\/[a-zA-Z0-9.+-]+(?:;codecs=[^;,]+)?;base64,/i,
    "",
  );
  const compact = stripped.replace(/\s/g, "");

  if (!compact || !/^[A-Za-z0-9+/]*={0,2}$/.test(compact)) {
    throw new Error("Invalid base64 audio data.");
  }

  const buffer = Buffer.from(compact, "base64");
  if (!buffer.length) throw new Error("Audio payload is empty.");

  const normalizedInput = compact.replace(/=+$/, "");
  const normalizedOutput = buffer.toString("base64").replace(/=+$/, "");
  if (normalizedInput !== normalizedOutput) {
    throw new Error("Malformed base64 audio data.");
  }
  return buffer;
}

      const parsed = TranscriptionRequestSchema.safeParse(req.body);
      if (!parsed.success) {
        res.status(400).json({
          error: "Invalid request.",
          details: parsed.error.flatten(),
        });
        return;

      const user = req.courtUser;

        res.status(401).json({ error: "Authentication required." });

        },
      });

      const responseText = response.text?.trim();
      if (!responseText) {
        res.status(502).json({ error: "Transcription provider returned no result." });
        return;
      }

      let raw: unknown;
      try {
        raw = JSON.parse(responseText);
      } catch {
        res.status(502).json({ error: "Transcription provider returned invalid JSON." });
        return;
      }

      const validated = TranscriptionResponseSchema.safeParse(raw);
      if (!validated.success) {
        // Do not return raw model output: it may contain confidential data.
        res.status(502).json({
          error: "Transcription result did not match the required schema.",
        });
        return;
      }

      // TODO: Write a metadata-only audit event using your approved audit service.
      // Never log audio or transcript text. Preserve original audio separately.
      res.status(200).json({
        ...validated.data,
        modelUsed: modelName,
        mimeType: normalizedMime,
        reviewRequired: true,
      });
    } catch (error) {
      // Log only a request ID and sanitized error category in production.
      const errorId = randomUUID();
      console.error("Transcription request failed. Error ID:", errorId);
      res.status(500).json({
        error: "Transcription failed. Please try again.",
        errorId,
      });
    }
  },
);

export default router;

import type { AudioRecordingResult } from "./services/CourtroomAudioService";

export interface TranscriptionRequestOptions {
  language?: "auto" | "om" | "am" | "en";
  verbatimMode?: boolean;
  caseId: string;
  firebaseIdToken: string;
}

export async function transcribeShortRecording(
  recording: AudioRecordingResult,
  options: TranscriptionRequestOptions,
): Promise<unknown> {
  if (recording.sizeBytes > 15 * 1024 * 1024) {
    throw new Error("Clip exceeds the short-recording limit.");
  }

  const base64 = await new Promise<string>((resolve, reject) => {
    const reader = new FileReader();
    reader.onerror = () => reject(new Error("Could not read audio file."));
    reader.onload = () => {
      const dataUrl = String(reader.result ?? "");
      const comma = dataUrl.indexOf(",");
      if (comma < 0) {
        reject(new Error("Invalid audio data URL."));
        return;
      }
      resolve(dataUrl.slice(comma + 1));
    };
    reader.readAsDataURL(recording.blob);
  });

  const response = await fetch("/api/transcribe", {
    method: "POST",
    headers: {
      "Content-Type": "application/json",
      Authorization: `Bearer ${options.firebaseIdToken}`,
    },
    body: JSON.stringify({
      audioData: base64,
      mimeType: recording.mimeType,
      language: options.language ?? "auto",
      verbatimMode: options.verbatimMode ?? true,
      caseId: options.caseId,
    }),
  });

  const body = await response.json().catch(() => ({}));
  if (!response.ok) {
    throw new Error(
      typeof body.error === "string"
        ? body.error
        : `Transcription failed (${response.status}).`,
    );
  }
  return body;
}
