} from ‘telegraf’; import { timingSafeEqual } from ‘node:crypto’;

TELEGRAM_WEBHOOK_SECRET = process.env.TELEGRAM_WEBHOOK_SECRET;

if (!TELEGRAM_BOT_TOKEN) { throw new Error(‘TELEGRAM_BOT_TOKEN is
missing’); }

if (!TELEGRAM_WEBHOOK_SECRET) { throw new Error(‘TELEGRAM_WEBHOOK_SECRET
is missing’); }

// ===================================================== // 2. LEGAL AI
SERVICES // =====================================================

// Functions kana project SEERAAI keessan keessatti // duraan jiran

import { retrieveLegalGrounding, generateAIAnswer } from

TRILINGUAL SYSTEM INSTRUCTION //

// ===================================================== // 4. EXPRESS
AND TELEGRAM BOT //
=====================================================

export const app = express();

app.use(express.json({ limit: ‘1mb’ }));

const bot = new Telegraf(TELEGRAM_BOT_TOKEN);

// ===================================================== // 5. WEBHOOK
SECRET VALIDATION //
=====================================================

function verifyWebhookSecret(received: string): boolean { if (!received
|| !TELEGRAM_WEBHOOK_SECRET) { return false; }

const receivedBuffer = Buffer.from(received); const expectedBuffer =
Buffer.from( TELEGRAM_WEBHOOK_SECRET );

if (receivedBuffer.length !== expectedBuffer.length) { return false; }

return timingSafeEqual(receivedBuffer, expectedBuffer); }

// ===================================================== // 6. TELEGRAM
MESSAGE CHUNKING //
=====================================================

function splitMessage( message: string, maxLength = 3500 ): string[] {
const chunks: string[] = []; let remaining = message.trim();

while (remaining.length > maxLength) { let splitAt =
remaining.lastIndexOf( ‘’, maxLength );

    if (splitAt < maxLength * 0.5) {
      splitAt = remaining.lastIndexOf(
        ' ',
        maxLength
      );
    }

    if (splitAt < 1) {
      splitAt = maxLength;
    }

    chunks.push(remaining.slice(0, splitAt));
    remaining = remaining.slice(splitAt).trim();

}

if (remaining.length > 0) { chunks.push(remaining); }

return chunks; }

// ===================================================== // 7. LEGAL
QUESTION HANDLER //
=====================================================

bot.on(‘text’, async (ctx) => { const userQuery =
ctx.message.text.trim();

if (!userQuery) { await ctx.reply(‘Maaloo gaaffii keessan barreessaa.’);
return; }

if (userQuery.length > 10000) { await ctx.reply( ‘Gaaffiin keessan
baayyee dheeraa dha.’ + ‘Maaloo gabaabsaa ergaa.’ ); return; }

try { await ctx.sendChatAction(‘typing’);

    // STEP 1: Retrieve legal documents

    const ragGrounding =
      await retrieveLegalGrounding(userQuery);

    // STEP 2: Generate grounded AI answer

    const aiResponse = await generateAIAnswer({
      prompt: userQuery,
      context: ragGrounding,
      systemInstruction:
        TRILINGUAL_SYSTEM_INSTRUCTION
    });

    const responseText =
      typeof aiResponse === 'string'
        ? aiResponse
        : aiResponse?.text;

    if (
      typeof responseText !== 'string' ||
      !responseText.trim()
    ) {
      throw new Error('AI returned an empty response');
    }

    // STEP 3: Split long responses

    const chunks = splitMessage(responseText);

    // STEP 4: Send answer to Telegram

    for (const chunk of chunks) {
      await ctx.reply(chunk);
    }

} catch (error) { console.error( ‘SEERAAI Legal AI Error:’, error );

    try {
      await ctx.reply(
        'Dhiifama, yeroo ammaa deebii seeraa ' +
        'qopheessuu hin dandeenye. Maaloo ' +
        'yeroo muraasa booda irra deebi\'aa.'
      );
    } catch (replyError) {
      console.error(
        'Telegram reply error:',
        replyError
      );
    }

} });

// ===================================================== // 8. NON-TEXT
MESSAGE HANDLER // =====================================================

bot.on(‘message’, async (ctx) => { await ctx.reply( ‘Maaloo gaaffii
seeraa keessan barreeffamaan ergaa.’ + ‘Deeggarsi suuraa fi sagalee yoo
barbaachise’ + ‘moduulii isaanii addatti walitti hidhaa.’ ); });

// ===================================================== // 9. TELEGRAM
WEBHOOK ENDPOINT //
=====================================================

app.post( ‘/api/telegram/webhook’, async (req: Request, res: Response)
=> { const receivedSecret = req.get(‘X-Telegram-Bot-Api-Secret-Token’)
|| ’’;

    if (!verifyWebhookSecret(receivedSecret)) {
      res.sendStatus(401);
      return;
    }

    try {
      await bot.handleUpdate(req.body);
      res.sendStatus(200);

    } catch (error) {
      console.error(
        'Telegram Webhook Error:',
        error
      );

      res.sendStatus(500);
    }

} );

// ===================================================== // 10. HEALTH
CHECK // =====================================================

app.get(‘/api/telegram/health’, (_req, res) => { res.status(200).json({
ok: true, service: ‘SEERAAI Telegram Bot’, mode: ‘webhook’, status:
‘running’ }); });

APP // =====================================================
